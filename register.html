<script>
const usernameInput = document.getElementById('username');
const codeInput = document.getElementById('registration_code');
const avatarPreview = document.getElementById('avatar-preview');
const habboStatus = document.getElementById('habbo-status');
const employeeInfo = document.getElementById('employee-info');
const passwordInput = document.getElementById('password');
const passwordConfirm = document.getElementById('password_confirm');
const registerSubmit = document.getElementById('registerSubmit');
let lookupTimer = null;
let canRegister = false;

const setCanRegister = (allowed) => {
canRegister = allowed;
codeInput.disabled = !allowed;
passwordInput.disabled = !allowed;
passwordConfirm.disabled = !allowed;
registerSubmit.disabled = !allowed;
};

usernameInput?.addEventListener('input', () => {
clearTimeout(lookupTimer);
lookupTimer = setTimeout(async () => {
const name = usernameInput.value.trim();
if (!name) {
avatarPreview.src = '/assets/img/avatar-placeholder.svg';
habboStatus.textContent = '';
employeeInfo.textContent = '';
setCanRegister(false);
return;
}
habboStatus.textContent = 'Prüfe Mitarbeiter…';
employeeInfo.textContent = '';
setCanRegister(false);
try {
const res = await fetch('/api/register-check?username=' + encodeURIComponent(name));
const data = await res.json();
if (data.ok && data.can_register) {
avatarPreview.src = data.habbo.avatar_head_url;
habboStatus.innerHTML = '<span class="text-success"><i class="bi bi-check-circle"></i> ' + data.habbo.name + ' – berechtigt für Tool-Zugang</span>';
if (data.employee.rank_name) {
employeeInfo.innerHTML = '<span class="text-muted">Agentur-Rang: <strong>' + data.employee.rank_name + '</strong>'
+ (data.employee.section_name ? ' (' + data.employee.section_name + ')' : '') + '</span>';
}
setCanRegister(true);
} else {
avatarPreview.src = '/assets/img/avatar-placeholder.svg';
habboStatus.innerHTML = '<span class="text-danger"><i class="bi bi-x-circle"></i> ' + (data.error || 'Nicht berechtigt') + '</span>';
if (data.employee && data.employee.rank_name) {
employeeInfo.innerHTML = '<span class="text-muted">Rang: ' + data.employee.rank_name + '</span>';
}
setCanRegister(false);
}
} catch (e) {
habboStatus.textContent = '';
setCanRegister(false);
}
}, 400);
});

document.getElementById('register-form')?.addEventListener('submit', (event) => {
if (!canRegister) event.preventDefault();
});

if (usernameInput.value.trim()) {
usernameInput.dispatchEvent(new Event('input'));
}
</script>
